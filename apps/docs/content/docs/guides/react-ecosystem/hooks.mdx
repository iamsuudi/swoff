## React Hooks

These hooks listen to window events dispatched by the client registration code and expose reactive state for your components.

## usePWAUpdate

Main hook for the update flow.

```typescript
const {
  currentVersion, // string | null — currently registered version
  availableVersion, // string | null — version available for update
  updateStatus, // 'idle' | 'available' | 'downloading'
  progress, // number (0-100)
  forceUpdate, // boolean — whether update is mandatory
  dismissUpdate, // () => void — dismiss (disabled if forceUpdate)
  acceptUpdate, // () => Promise<void> — download and activate
  checkForUpdates, // () => Promise<void> — manually check version.json
} = usePWAUpdate();
```

**Usage:**

```typescript
function App() {
  const { updateStatus, acceptUpdate } = usePWAUpdate();
  if (updateStatus === 'available') {
    return <button onClick={acceptUpdate}>Update</button>;
  }
}
```

**Implementation:**

```typescript
import { useState, useEffect, useCallback } from "react";

export function usePWAUpdate() {
  const [updateStatus, setUpdateStatus] = useState<
    "idle" | "available" | "downloading"
  >("idle");
  const [currentVersion, setCurrentVersion] = useState<string | null>(null);
  const [availableVersion, setAvailableVersion] = useState<string | null>(null);
  const [progress, setProgress] = useState(0);
  const [forceUpdate, setForceUpdate] = useState(false);

  useEffect(() => {
    const handleUpdateAvailable = (e: CustomEvent) => {
      const newVersion = e.detail.version;
      const current = localStorage.getItem("swRegisteredVersion");
      setCurrentVersion(current);
      setAvailableVersion(newVersion);
      setUpdateStatus("available");
      const minSupported = window.swMinSupportedVersion || "0.0.0";
      setForceUpdate(current < minSupported);
    };

    const handleProgress = (e: CustomEvent) => {
      if (updateStatus === "downloading") setProgress(e.detail.percent);
    };

    const handleReady = () => {
      if (updateStatus === "downloading") {
        setUpdateStatus("idle");
        setProgress(0);
      }
    };

    window.addEventListener(
      "sw-update-available",
      handleUpdateAvailable as any,
    );
    window.addEventListener("sw-progress", handleProgress as any);
    window.addEventListener("sw-ready", handleReady);
    return () => {
      window.removeEventListener(
        "sw-update-available",
        handleUpdateAvailable as any,
      );
      window.removeEventListener("sw-progress", handleProgress as any);
      window.removeEventListener("sw-ready", handleReady);
    };
  }, [updateStatus]);

  const acceptUpdate = useCallback(async () => {
    setUpdateStatus("downloading");
    await registerServiceWorker(availableVersion);
  }, [availableVersion]);

  const dismissUpdate = useCallback(() => {
    if (!forceUpdate) {
      setUpdateStatus("idle");
      sessionStorage.setItem("sw-dismissed-update", "true");
    }
  }, [forceUpdate]);

  const checkForUpdates = useCallback(async () => {
    const manifest = await checkForUpdate();
  }, []);

  return {
    currentVersion,
    availableVersion,
    updateStatus,
    progress,
    forceUpdate,
    dismissUpdate,
    acceptUpdate,
    checkForUpdates,
  };
}
```

## useSWProgress

Track SW download progress.

```typescript
const { progress, status } = useSWProgress();
// status: 'idle' | 'installing' | 'ready' | 'error'
```

**Usage:**

```typescript
function ProgressBar() {
  const { progress, status } = useSWProgress();
  if (status !== 'installing') return null;
  return <div style={{ width: `${progress}%` }}>...</div>;
}
```

**Implementation:**

```typescript
import { useState, useEffect } from "react";

export function useSWProgress() {
  const [progress, setProgress] = useState(0);
  const [status, setStatus] = useState<
    "idle" | "installing" | "ready" | "error"
  >("idle");

  useEffect(() => {
    const handleProgress = (e: CustomEvent) => {
      setStatus("installing");
      setProgress(e.detail.percent);
    };

    const handleReady = () => {
      setStatus("ready");
      setTimeout(() => setStatus("idle"), 1000);
    };

    const handleError = () => setStatus("error");

    window.addEventListener("sw-progress", handleProgress as any);
    window.addEventListener("sw-ready", handleReady);
    window.addEventListener("sw-error", handleError);
    return () => {
      window.removeEventListener("sw-progress", handleProgress as any);
      window.removeEventListener("sw-ready", handleReady);
      window.removeEventListener("sw-error", handleError);
    };
  }, []);

  return { progress, status };
}
```

## usePWAVersion

Read current/latest versions.

```typescript
const { currentVersion, latestVersion } = usePWAVersion();
```

**Implementation:**

```typescript
import { useState, useEffect } from "react";

export function usePWAVersion() {
  const [currentVersion, setCurrentVersion] = useState<string | null>(null);
  const [latestVersion, setLatestVersion] = useState<string | null>(null);

  useEffect(() => {
    const handleDetected = () => {
      setCurrentVersion(window.currentSWVersion || null);
      setLatestVersion(window.latestSWVersion || null);
    };

    window.addEventListener("sw-version-detected", handleDetected);
    handleDetected();
    return () =>
      window.removeEventListener("sw-version-detected", handleDetected);
  }, []);

  return { currentVersion, latestVersion };
}
```

## useCrossTabCache

Reactively respond to cache invalidations triggered by other tabs.

```typescript
const { invalidatedTags } = useCrossTabCache();
// invalidatedTags: string[] — tag names invalidated by another tab (clears after tick)
```

**Usage:**

```typescript
function TodoList() {
  const { invalidatedTags } = useCrossTabCache();
  const [todos, setTodos] = useState([]);

  useEffect(() => {
    if (invalidatedTags.includes("todos")) {
      fetchWithCache("/api/todos", { tags: ["todos"] })
        .then((r) => r.json())
        .then(setTodos);
    }
  }, [invalidatedTags]);

  // ...
}
```

**Implementation:**

```typescript
import { useState, useEffect } from "react";

export function useCrossTabCache() {
  const [invalidatedTags, setInvalidatedTags] = useState<string[]>([]);

  useEffect(() => {
    function handleInvalidation(e: CustomEvent) {
      setInvalidatedTags(e.detail.tags);
      setTimeout(() => setInvalidatedTags([]), 0);
    }

    window.addEventListener(
      "cache-invalidated",
      handleInvalidation as EventListener,
    );
    return () =>
      window.removeEventListener(
        "cache-invalidated",
        handleInvalidation as EventListener,
      );
  }, []);

  return { invalidatedTags };
}
```

## useOfflineStatus

Track network connectivity state and show appropriate UI.

```typescript
const { isOnline, wasOffline } = useOfflineStatus();
// isOnline: boolean — current connectivity
// wasOffline: boolean — user was offline since last render
```

**Usage:**

```typescript
function OfflineBanner() {
  const { isOnline } = useOfflineStatus();
  if (isOnline) return null;
  return <div className="banner">You're offline</div>;
}
```

**Implementation:**

```typescript
import { useState, useEffect } from "react";

export function useOfflineStatus() {
  const [isOnline, setIsOnline] = useState(navigator.onLine);
  const [wasOffline, setWasOffline] = useState(false);

  useEffect(() => {
    const handleOnline = () => {
      setWasOffline(!isOnline);
      setIsOnline(true);
    };
    const handleOffline = () => {
      setWasOffline(true);
      setIsOnline(false);
    };

    window.addEventListener("online", handleOnline);
    window.addEventListener("offline", handleOffline);

    return () => {
      window.removeEventListener("online", handleOnline);
      window.removeEventListener("offline", handleOffline);
    };
  }, [isOnline]);

  return { isOnline, wasOffline };
}
```

## useMutationQueue

React hook for offline mutation handling with queue state.

```typescript
const { pendingCount, isSyncing, lastSync, retryMutation } = useMutationQueue();
// pendingCount: number — mutations waiting to sync
// isSyncing: boolean — currently processing queue
// lastSync: { succeeded: number, failed: number } | null
// retryMutation: () => void — manually retry failed mutations
```

**Usage:**

```typescript
function SyncIndicator() {
  const { pendingCount, isSyncing } = useMutationQueue();

  if (pendingCount === 0) return null;

  return (
    <div className="sync-indicator">
      {isSyncing ? "Syncing..." : `${pendingCount} pending`}
    </div>
  );
}
```

**Implementation:**

```typescript
import { useState, useEffect, useCallback } from "react";
import { getPendingCount, processMutationQueue } from "../../../swoff/mutation-queue.js";

export function useMutationQueue() {
  const [pendingCount, setPendingCount] = useState(0);
  const [isSyncing, setIsSyncing] = useState(false);
  const [lastSync, setLastSync] = useState<{ succeeded: number; failed: number } | null>(null);

  const updateCount = useCallback(async () => {
    setPendingCount(await getPendingCount());
  }, []);

  useEffect(() => {
    updateCount();

    const onQueueChange = () => updateCount();
    const onSyncComplete = (e: CustomEvent) => {
      setIsSyncing(false);
      setLastSync(e.detail);
      updateCount();
    };
    const onOnline = () => {
      setIsSyncing(true);
      processMutationQueue();
    };

    window.addEventListener("mutation-queue-changed", onQueueChange);
    window.addEventListener("mutation-sync-complete", onSyncComplete as EventListener);
    window.addEventListener("online", onOnline);

    return () => {
      window.removeEventListener("mutation-queue-changed", onQueueChange);
      window.removeEventListener("mutation-sync-complete", onSyncComplete as EventListener);
      window.removeEventListener("online", onOnline);
    };
  }, [updateCount]);

  const retryMutation = useCallback(async () => {
    setIsSyncing(true);
    await processMutationQueue();
  }, []);

  return { pendingCount, isSyncing, lastSync, retryMutation };
}
```

## useOfflineData

Fetch data that works offline with automatic caching.

```typescript
const { data, isLoading, isOffline, refetch } = useOfflineData("/api/todos", {
  tags: ["todos"],
});
// data: T | null
// isLoading: boolean
// isOffline: boolean — currently serving cached data
// refetch: () => void
```

**Implementation:**

```typescript
import { useState, useEffect, useCallback } from "react";
import { fetchWithCache } from "../../../swoff/fetch-wrapper.js";
import { useOfflineStatus } from "./useOfflineStatus";

export function useOfflineData<T>(url: string, options: { tags?: string[] } = {}) {
  const [data, setData] = useState<T | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [isOffline, setIsOffline] = useState(false);
  const { isOnline } = useOfflineStatus();

  const refetch = useCallback(async () => {
    setIsLoading(true);
    try {
      const response = await fetchWithCache(url, options);
      const json = await response.json();
      setData(json);
      setIsOffline(!isOnline);
    } catch (error) {
      console.error("Failed to fetch:", error);
    } finally {
      setIsLoading(false);
    }
  }, [url, JSON.stringify(options), isOnline]);

  useEffect(() => {
    refetch();
  }, [refetch]);

  return { data, isLoading, isOffline, refetch };
}
```
