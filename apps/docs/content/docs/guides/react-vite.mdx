# React + Vite Configuration

This guide shows you how to configure Swoff for React applications using Vite. The CLI makes it easy to set up offline functionality in seconds.

## Step 1: Install Swoff CLI

```bash
npm install -D @swoff/cli
```

## Step 2: Initialize Swoff

Run the CLI to initialize Swoff with React + Vite defaults:

```bash
swoff init --framework react-vite
```

This creates:
- `swoff.config.json` with React + Vite optimized settings
- `src/hooks/`, `src/components/`, `src/utils/` directories

## Step 5: Register Service Worker

Create `src/utils/sw-registration.ts`:

```typescript
export const registerServiceWorker = () => {
  if ('serviceWorker' in navigator) {
    const version = import.meta.env.VITE_APP_VERSION || '1.0.0';
    navigator.serviceWorker
      .register(`/sw-v${version}.js`)
      .then((registration) => {
        console.log('Service Worker registered:', registration);
        
        // Handle updates
        registration.addEventListener('updatefound', () => {
          const installingWorker = registration.installing;
          installingWorker?.addEventListener('statechange', () => {
            if (installingWorker.state === 'installed' && navigator.serviceWorker.controller) {
              console.log('New version available');
              // Show update notification
            }
          });
        });
      })
      .catch((error) => {
        console.error('Service Worker registration failed:', error);
      });
  }
};
```

## Step 6: Create Offline Hook

Create `src/hooks/useOffline.ts`:

```typescript
import { useState, useEffect } from 'react';

export const useOffline = () => {
  const [isOnline, setIsOnline] = useState(navigator.onLine);

  useEffect(() => {
    const handleOnline = () => setIsOnline(true);
    const handleOffline = () => setIsOnline(false);

    window.addEventListener('online', handleOnline);
    window.addEventListener('offline', handleOffline);

    return () => {
      window.removeEventListener('online', handleOnline);
      window.removeEventListener('offline', handleOffline);
    };
  }, []);

  return { isOnline, isOffline: () => !isOnline };
};
```

## Step 8: Create App Store with Zustand

import { create } from 'zustand';

interface AppState {
  isOnline: boolean;
  isLoading: boolean;
  error: string | null;
  setOnline: (status: boolean) => void;
  setLoading: (loading: boolean) => void;
  setError: (error: string | null) => void;
}

export const useAppStore = create<AppState>((set) => ({
  isOnline: true,
  isLoading: false,
  error: null,
  setOnline: (status) => set({ isOnline: status }),
  setLoading: (loading) => set({ isLoading: loading }),
  setError: (error) => set({ error }),
}));

## Step 9: Create API Hook with Offline Support

Create `src/hooks/useApi.ts`:

```typescript
import { useState } from 'react';
import { useAppStore } from '../stores/appStore';

export const useApi = (endpoint: string) => {
  const appStore = useAppStore();
  const [data, setData] = useState<any>(null);
  const [error, setError] = useState<string | null>(null);

  const fetchData = async (options?: RequestInit) => {
    try {
      appStore.setLoading(true);
      appStore.setError(null);

      const response = await fetch(endpoint, {
        ...options,
        headers: {
          'Content-Type': 'application/json',
          ...options?.headers
        }
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const result = await response.json();
      setData(result);
      return result;
    } catch (err) {
      const errorMessage = err instanceof Error ? err.message : 'An error occurred';
      setError(errorMessage);
      appStore.setError(errorMessage);
      
      // If offline and we have cached data, return it
      if (!navigator.onLine && data) {
        console.log('Using cached data while offline');
        return data;
      }
      
      throw err;
    } finally {
      appStore.setLoading(false);
    }
  };

  return {
    data,
    error,
    fetchData,
    isLoading: appStore.isLoading
  };
};
```

Load the appropriate config in `vite.config.ts`:

```typescript
import { loadEnv } from 'vite';

export default ({ mode }) => {
  const env = loadEnv(mode, process.cwd());
  
  const config = mode === 'development' 
    ? require('./swoff.config.development.json')
    : require('./swoff.config.production.json');

  return {
    // ... other config
    define: {
      'import.meta.env.SWOFF_CONFIG': JSON.stringify(config)
    }
  };
};
```

### Custom Error Boundary

Create `src/components/ErrorBoundary.tsx`:

```typescript
import React, { Component, ReactNode, ErrorInfo } from 'react';

interface Props {
  children: ReactNode;
}

interface State {
  hasError: boolean;
  error: Error | null;
}

export class ErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props);
    this.state = { hasError: false, error: null };
  }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error('Error caught by boundary:', error, errorInfo);
  }

  render() {
    if (this.state.hasError) {
      return (
        <div className="min-h-screen flex items-center justify-center">
          <div className="text-center">
            <h2 className="text-2xl font-bold mb-4">Something went wrong</h2>
            <p className="text-gray-600 mb-4">
              {this.state.error?.message || 'An unexpected error occurred'}
            </p>
            <button 
              onClick={() => window.location.reload()}
              className="px-4 py-2 bg-blue-500 text-white rounded hover:bg-blue-600"
            >
              Retry
            </button>
          </div>
        </div>
      );
    }

    return this.props.children;
  }
}
```

### Offline Form Component

Create `src/components/OfflineForm.tsx`:

```typescript
import React, { useState } from 'react';
import { useOffline } from '../hooks/useOffline';
import { useApi } from '../hooks/useApi';

interface FormData {
  title: string;
  content: string;
}

interface OfflineFormProps {
  endpoint: string;
}

export const OfflineForm: React.FC<OfflineFormProps> = ({ endpoint }) => {
  const { isOffline } = useOffline();
  const { data, error, isLoading, fetchData } = useApi(endpoint);
  const [formData, setFormData] = useState<FormData>({ title: '', content: '' });

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    
    try {
      await fetchData({
        method: 'POST',
        body: JSON.stringify(formData)
      });
      
      // Reset form on success
      setFormData({ title: '', content: '' });
    } catch (err) {
      console.error('Failed to save:', err);
    }
  };

  const handleChange = (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => {
    const { name, value } = e.target;
    setFormData(prev => ({ ...prev, [name]: value }));
  };

  return (
    <div className="max-w-md mx-auto">
      <form onSubmit={handleSubmit} className="space-y-4">
        <div>
          <label htmlFor="title" className="block text-sm font-medium text-gray-700">
            Title
          </label>
          <input
            id="title"
            name="title"
            type="text"
            value={formData.title}
            onChange={handleChange}
            disabled={isLoading || isOffline}
            className="mt-1 block w-full px-3 py-2 border border-gray-300 rounded-md shadow-sm focus:outline-none focus:ring-blue-500 focus:border-blue-500 disabled:bg-gray-100"
          />
        </div>
        
        <div>
          <label htmlFor="content" className="block text-sm font-medium text-gray-700">
            Content
          </label>
          <textarea
            id="content"
            name="content"
            value={formData.content}
            onChange={handleChange}
            disabled={isLoading || isOffline}
            rows={4}
            className="mt-1 block w-full px-3 py-2 border border-gray-300 rounded-md shadow-sm focus:outline-none focus:ring-blue-500 focus:border-blue-500 disabled:bg-gray-100"
          />
        </div>
        
        <button
          type="submit"
          disabled={isLoading || isOffline}
          className="w-full flex justify-center py-2 px-4 border border-transparent rounded-md shadow-sm text-sm font-medium text-white bg-blue-600 hover:bg-blue-700 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-blue-500 disabled:bg-gray-400"
        >
          {isLoading ? 'Saving...' : 'Save'}
        </button>
      </form>
      
      {isOffline && (
        <div className="mt-4 p-4 bg-yellow-50 border border-yellow-200 rounded-md">
          <p className="text-sm text-yellow-800">
            You are offline. Your changes will be saved when you're back online.
          </p>
        </div>
      )}
      
      {error && (
        <div className="mt-4 p-4 bg-red-50 border border-red-200 rounded-md">
          <p className="text-sm text-red-800">
            Error: {error}
          </p>
        </div>
      )}
    </div>
  );
};
```

// src/hooks/useOffline.test.ts
import { renderHook, act } from '@testing-library/react';
import { useOffline } from './useOffline';

describe('useOffline', () => {
  it('should detect online status', () => {
    const { result } = renderHook(() => useOffline());
    
    expect(result.current.isOnline).toBe(true);
    expect(result.current.isOffline()).toBe(false);
  });

  it('should handle offline events', () => {
    const { result } = renderHook(() => useOffline());
    
    act(() => {
      window.dispatchEvent(new Event('offline'));
    });
    
    expect(result.current.isOnline).toBe(false);
    expect(result.current.isOffline()).toBe(true);
  });
});

## Troubleshooting

### Common Issues

1. **Service Worker not registering**
   - Check file path and permissions
   - Verify HTTPS in production
   - Check browser console for errors

2. **Cache not working**
   - Verify cache strategies match your URLs
   - Check that assets are being requested correctly
   - Use DevTools to inspect cache contents

3. **PWA not installing**
   - Verify manifest.json is properly formatted
   - Check that all required icons are present
   - Test on different browsers

### Debugging Tools

- **React DevTools**: Inspect component state
- **Chrome DevTools**: Service Workers tab for registration and events
- **Lighthouse**: Run PWA audits
- **Vite DevTools**: Monitor build and development process

This configuration provides a solid foundation for offline-first React applications with Vite. Adjust the strategies and settings based on your specific application needs.
