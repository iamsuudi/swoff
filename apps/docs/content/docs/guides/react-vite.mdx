# React + Vite

Quick setup for **React + Vite** projects.

## Create Project

```bash
npm create vite@latest my-app -- --template react-ts
cd my-app && npm install
```

## Project Structure

```tsx title="Docu" ts2js
import { ReactNode } from "react";

export default function Layout({ children }: { children: ReactNode }) {
  return <div>{children}</div>;
}
```

```
my-app
├── src
│   ├── main.tsx
│   ├── App.tsx
│   └── sw-template.js
├── generate-sw.js
├── sw.d.ts
└── package.json
```

## Install Dependencies

```bash
npm install react@latest react-dom@latest
npm install -D typescript @types/react vite @vitejs/plugin-react
```

## Update main.tsx

```typescript
import React from 'react'
import ReactDOM from 'react-dom/client'
import App from './App'

const STORAGE_KEY = 'swRegisteredVersion'

async function registerSW() {
  const { version } = await fetch('/version.json').then(r => r.json())
  await navigator.serviceWorker.register(`/sw-v${version}.js`)
  localStorage.setItem(STORAGE_KEY, version)
}

if ('serviceWorker' in navigator) {
  window.addEventListener('load', registerSW)
}

ReactDOM.createRoot(document.getElementById('root')!).render(<App />)
```

## Update package.json

```json
{
  "scripts": {
    "build": "vite build && node generate-sw.js"
  }
}
```

## Build & Test

```bash
npm version patch
npm run build
```

Test offline: DevTools → Application → Service Workers → Check "Offline".
