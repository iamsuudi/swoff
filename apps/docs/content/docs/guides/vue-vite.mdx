# Vue + Vite Configuration

This guide shows you how to configure Swoff for Vue applications using Vite. The configuration-driven approach makes it easy to set up offline functionality without writing custom service worker code.

## Step 1: Install Dependencies

```bash
npm install
```

## Step 2: Create Swoff Configuration

Create `swoff.config.json` in your project root:

```json
{
  "$schema": "https://swoff.netlify.app/schema/v1.json",
  "enabled": true,
  "version": "from-package",
  "minSupportedVersion": "1.0.0",
  "serviceWorker": {
    "autoUpdate": false,
    "defaultStrategy": "cache-first",
    "strategies": {
      "/api/*": "network-first",
      "/assets/*": "cache-first",
      "/images/*": "stale-while-revalidate"
    },
    "maxCacheEntries": 150,
    "maxCacheAge": 43200000
  },
  "features": {
    "offlineReads": true,
    "mutationQueue": true,
    "pwa": true,
    "crossTabSync": true,
    "tagInvalidation": true
  },
  "database": {
    "name": "vue-app-db",
    "stores": ["users", "posts", "settings"]
  },
  "build": {
    "outputDir": "dist",
    "swFilename": "sw"
  }
}
```

### Configuration Notes for Vue + Vite:

- **`/assets/*`**: Cache Vue static assets
- **`/api/*`**: Use network-first for API calls
- **`/images/*`**: Use stale-while-revalidate for images
- **`build.outputDir: "dist"`**: Output service worker to dist directory
- **`build.swFilename: "sw"`**: Service worker accessible at `/sw-v{version}.js`

## Step 6: Register Service Worker

Create `src/utils/sw-registration.ts`:

```typescript
export const registerServiceWorker = () => {
  if ('serviceWorker' in navigator) {
    const version = import.meta.env.VITE_APP_VERSION || '1.0.0';
    navigator.serviceWorker
      .register(`/sw-v${version}.js`)
      .then((registration) => {
        console.log('Service Worker registered:', registration);
        
        // Handle updates
        registration.addEventListener('updatefound', () => {
          const installingWorker = registration.installing;
          installingWorker?.addEventListener('statechange', () => {
            if (installingWorker.state === 'installed' && navigator.serviceWorker.controller) {
              console.log('New version available');
              // Show update notification
            }
          });
        });
      })
      .catch((error) => {
        console.error('Service Worker registration failed:', error);
      });
  }
};
```

## Step 7: Create Composable for Offline Detection

Create `src/composables/useOffline.ts`:

```typescript
import { ref, onMounted, onUnmounted } from 'vue';

export const useOffline = () => {
  const isOnline = ref(navigator.onLine);

  const updateOnlineStatus = () => {
    isOnline.value = navigator.onLine;
  };

  const updateOfflineStatus = () => {
    isOnline.value = false;
  };

  onMounted(() => {
    window.addEventListener('online', updateOnlineStatus);
    window.addEventListener('offline', updateOfflineStatus);
  });

  onUnmounted(() => {
    window.removeEventListener('online', updateOnlineStatus);
    window.removeEventListener('offline', updateOfflineStatus);
  });

  return {
    isOnline,
    isOffline: () => !isOnline.value
  };
};
```

## Step 9: Create Pinia Store for App State

Create `src/stores/appStore.ts`:

```typescript
import { defineStore } from 'pinia';
import { ref } from 'vue';

export const useAppStore = defineStore('app', () => {
  const isOnline = ref(true);
  const isLoading = ref(false);
  const error = ref<string | null>(null);

  const setOnline = (status: boolean) => {
    isOnline.value = status;
  };

  const setLoading = (loading: boolean) => {
    isLoading.value = loading;
  };

  const setError = (err: string | null) => {
    error.value = err;
  };

  return {
    isOnline,
    isLoading,
    error,
    setOnline,
    setLoading,
    setError
  };
});
```

## Step 10: Create Data Fetching Composable

Create `src/composables/useApi.ts`:

```typescript
import { ref } from 'vue';
import { useAppStore } from '@/stores/appStore';

export const useApi = (endpoint: string) => {
  const appStore = useAppStore();
  const data = ref<any>(null);
  const error = ref<string | null>(null);

  const fetchData = async (options?: RequestInit) => {
    try {
      appStore.setLoading(true);
      appStore.setError(null);

      const response = await fetch(endpoint, {
        ...options,
        headers: {
          'Content-Type': 'application/json',
          ...options?.headers
        }
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const result = await response.json();
      data.value = result;
      return result;
    } catch (err) {
      const errorMessage = err instanceof Error ? err.message : 'An error occurred';
      error.value = errorMessage;
      appStore.setError(errorMessage);
      
      // If offline and we have cached data, return it
      if (!navigator.onLine && data.value) {
        console.log('Using cached data while offline');
        return data.value;
      }
      
      throw err;
    } finally {
      appStore.setLoading(false);
    }
  };

  return {
    data,
    error,
    fetchData,
    isLoading: appStore.isLoading
  };
};
```

## Advanced Configuration

### Environment-Specific Configs

Create different configs for different environments:

```json
// swoff.config.development.json
{
  "enabled": true,
  "serviceWorker": {
    "defaultStrategy": "network-first",
    "maxCacheAge": 1800000
  }
}

// swoff.config.production.json
{
  "enabled": true,
  "serviceWorker": {
    "defaultStrategy": "cache-first",
    "maxCacheAge": 86400000
  }
}
```

Load the appropriate config:

```typescript
// vite.config.ts
import { loadEnv } from 'vite';

export default ({ mode }) => {
  const env = loadEnv(mode, process.cwd());
  
  const config = mode === 'development' 
    ? require('./swoff.config.development.json')
    : require('./swoff.config.production.json');

  return {
    // ... other config
    define: {
      'import.meta.env.SWOFF_CONFIG': JSON.stringify(config)
    }
  };
};
```

### Custom Vue Components for Offline Support

Create a component that handles offline forms:

```vue
<template>
  <div class="offline-form">
    <form @submit.prevent="handleSubmit">
      <input 
        v-model="formData.title" 
        placeholder="Title" 
        :disabled="isLoading || isOffline"
      />
      <textarea 
        v-model="formData.content" 
        placeholder="Content" 
        :disabled="isLoading || isOffline"
      ></textarea>
      <button 
        type="submit" 
        :disabled="isLoading || isOffline"
      >
        {{ isLoading ? 'Saving...' : 'Save' }}
      </button>
    </form>
    
    <div v-if="isOffline && !hasInternet" class="offline-message">
      <p>You are offline. Your changes will be saved when you're back online.</p>
    </div>
    
    <div v-if="error" class="error-message">
      {{ error }}
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, computed } from 'vue';
import { useOffline } from '@/composables/useOffline';
import { useApi } from '@/composables/useApi';

const props = defineProps<{
  endpoint: string;
}>();

const { isOffline } = useOffline();
const { data, error, isLoading, fetchData } = useApi(props.endpoint);

const formData = ref({
  title: '',
  content: ''
});

const hasInternet = computed(() => !isOffline.value);

const handleSubmit = async () => {
  try {
    await fetchData({
      method: 'POST',
      body: JSON.stringify(formData.value)
    });
    
    // Reset form on success
    formData.value = { title: '', content: '' };
  } catch (err) {
    console.error('Failed to save:', err);
  }
};
</script>
```

## Troubleshooting

### Common Issues

1. **Service Worker not registering**
   - Check file path and permissions
   - Verify HTTPS in production
   - Check browser console for errors

2. **Cache not working**
   - Verify cache strategies match your URLs
   - Check that assets are being requested correctly
   - Use DevTools to inspect cache contents

3. **PWA not installing**
   - Verify manifest.json is properly formatted
   - Check that all required icons are present
   - Test on different browsers

### Debugging Tools

- **Vue DevTools**: Inspect component state
- **Chrome DevTools**: Service Workers tab for registration and events
- **Lighthouse**: Run PWA audits
- **Vite DevTools**: Monitor build and development process

This configuration provides a solid foundation for offline-first Vue applications with Vite. Adjust the strategies and settings based on your specific application needs.
